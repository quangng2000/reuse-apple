cmake_minimum_required(VERSION 3.22)
project(reuse_streaming_vulkan LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  find_library(VULKAN_LIBRARY vulkan REQUIRED)
  set(VULKAN_INCLUDE_DIR "")
else()
  find_path(VULKAN_INCLUDE_DIR vulkan/vulkan.h HINTS /opt/homebrew/include REQUIRED)
  find_library(VULKAN_LIBRARY vulkan HINTS /opt/homebrew/lib REQUIRED)
endif()
add_library(reuse_vulkan_core STATIC runtime.cpp)
target_include_directories(reuse_vulkan_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${VULKAN_INCLUDE_DIR})
target_link_libraries(reuse_vulkan_core PUBLIC ${VULKAN_LIBRARY})
target_compile_options(reuse_vulkan_core PRIVATE -O3 -fno-fast-math -ffp-contract=off -Wall -Wextra -Werror -Wno-missing-field-initializers)
add_executable(reuse-vulkan-cli cli.cpp)
target_link_libraries(reuse-vulkan-cli PRIVATE reuse_vulkan_core)
if(ANDROID)
  add_library(reuse_streaming_vulkan SHARED jni.cpp)
  target_link_libraries(reuse_streaming_vulkan PRIVATE reuse_vulkan_core)
else()
  find_package(JNI QUIET)
  if(NOT JNI_FOUND)
    find_path(JNI_INCLUDE_PATH jni.h HINTS /opt/homebrew/opt/openjdk@21/libexec/openjdk.jdk/Contents/Home/include)
    if(JNI_INCLUDE_PATH)
      set(JNI_FOUND TRUE)
      set(JNI_INCLUDE_DIRS ${JNI_INCLUDE_PATH} ${JNI_INCLUDE_PATH}/darwin)
    endif()
  endif()
  if(JNI_FOUND)
    add_library(reuse_streaming_vulkan SHARED jni.cpp)
    target_include_directories(reuse_streaming_vulkan PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(reuse_streaming_vulkan PRIVATE reuse_vulkan_core)
  endif()
endif()
if(TARGET reuse_streaming_vulkan)
  target_compile_options(reuse_streaming_vulkan PRIVATE -O3 -fno-fast-math -ffp-contract=off -Wall -Wextra -Werror -Wno-missing-field-initializers)
endif()
